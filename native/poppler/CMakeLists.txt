cmake_minimum_required(VERSION 3.20)
project(tpe_poppler_provider LANGUAGES CXX)

# Deliberately opt-in: no downloading, bundling, or default system resolution.
set(POPPLER_INCLUDE_DIR "" CACHE PATH "Poppler 26.09.0 include/poppler directory")
set(POPPLER_LIBRARY "" CACHE FILEPATH "Exact Poppler 26.09.0 core shared library")
if(NOT IS_ABSOLUTE "${POPPLER_INCLUDE_DIR}" OR NOT IS_DIRECTORY "${POPPLER_INCLUDE_DIR}")
  message(FATAL_ERROR "Set POPPLER_INCLUDE_DIR to the absolute installed include/poppler directory")
endif()
if(NOT IS_ABSOLUTE "${POPPLER_LIBRARY}" OR NOT EXISTS "${POPPLER_LIBRARY}")
  message(FATAL_ERROR "Set POPPLER_LIBRARY to the absolute core shared library file")
endif()
add_library(tpe-poppler-provider SHARED provider.cc)
target_compile_features(tpe-poppler-provider PRIVATE cxx_std_20)
target_include_directories(tpe-poppler-provider SYSTEM PRIVATE "${POPPLER_INCLUDE_DIR}")
target_link_libraries(tpe-poppler-provider PRIVATE "${POPPLER_LIBRARY}")
if(MSVC)
  set_target_properties(tpe-poppler-provider PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_compile_options(tpe-poppler-provider PRIVATE /W4 /WX)
else()
  target_compile_options(tpe-poppler-provider PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set_target_properties(tpe-poppler-provider PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME poppler_provider_contract COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_provider.py"
    "$<TARGET_FILE:tpe-poppler-provider>" "${POPPLER_LIBRARY}")
endif()
